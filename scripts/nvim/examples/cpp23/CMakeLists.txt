cmake_minimum_required(VERSION 3.21)
project(NvimCppSmoke LANGUAGES CXX)
add_executable(nvim_cpp_smoke src/main.cpp)
target_compile_features(nvim_cpp_smoke PRIVATE cxx_std_23)
set_target_properties(nvim_cpp_smoke PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC OR CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(nvim_cpp_smoke PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME cpp23_smoke COMMAND nvim_cpp_smoke)
